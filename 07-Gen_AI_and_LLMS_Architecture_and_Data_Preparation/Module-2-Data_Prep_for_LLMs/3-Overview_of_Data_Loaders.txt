# Collate Function - used to keep og data untouched 
    apply data transformation, tokenization, converting to indices, transforming into tensor
    without modifying og data 

def collate_fn(batch):
    tensor_batch = []
    for sample in batch:
        tokens = tokenizer(sample)
        tensor_batch.append(torch.tensor([vocab[token] for token in tokens]))
    padded_batch = pad_sequence(tensor_batch, batch_first=True)
    return padded_batch
