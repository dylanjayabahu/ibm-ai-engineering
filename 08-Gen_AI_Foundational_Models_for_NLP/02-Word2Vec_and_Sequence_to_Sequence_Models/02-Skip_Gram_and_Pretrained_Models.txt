Skip-Gram 
    reverse of CBOW odel 
    predicts context words from target word 

    e.g. if context is 1 
        skip gram tries to predict word t-1 and t+1 given word t 
    
    the architecture is kind of the reverse of CBOW 

    input is of dim vocab_size, which is 1 hot encoded target word 
    hidden layer (embedding lookup)
    output is a vector of dim vocab_size, 
        representing one hots (softmax) over the vocab as the target word 
    
    we get 2*c training examples this way (we predict 1 context word at a time)
        e.g. if sentence is a b c d e f g 
        and c = 2
        and target = c 

        then training pairs are:
            c, a 
            c, b 
            c, d 
            c, e


class SkipGram_Model(nn.Module):
    def __init__(self, vocab_size, embed_dim):
        super (SkipGram_Model, self).__init__()

        self.embeddings = nn.Embedding(num_embeddings=vocab_size, embedding_dim=embed_dim)
        self.fc = nn.Linear(in_features=embed_dim, out_features=vocab_size)
    
    def forward(self, text):
        out = self.embeddings(text)
        out = torch.relu(out)
        out = self.fc(out)
        return out

def train_model(model, dataloader, criterion, optimizer, epochs=1000):
    epoch_losses=[]

    for epoch in tqdm(range(epochs)):
        running_loss = 0.0

        for idx, samples in enumerate(dataloader);
            optimizer.zero_grad()

            ## CBOW
            if any(isinstance(module, nn.EmbeddingBag)) for _, module in model.named_modules(): 
                target, context, offsets = samples 
                predicted = model(context)

            
            ## Skipgram
            elif any(isinstance(module, nn.Embedding) for _, module in model.named_modules()):
                target, context = samples 
                predicted = model(context)
            
            loss = criterion(predicted, target)
            loss.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), 0.1)
            optimizer.step()
            running_loss += loss.item()
        
        epoch_losses.append(running_loss / len(dataloader))
    return model, epoch_losses

class TextClassificationModel(nn.Module):
    def __init__(self, vocab_size, embed_dim, num_class):
        super(TextClassificationModel, self).__init__()
        self.embedding = torch.nn.Embedding.from_pretrained(glove_vetors_6b.vectors, freeze=True)
        self.fc = nn.Linear(embed_dim, num_class)
        self.init_weights()
    
    
    def init_weights(self):
        initrange = 0.5
        self.fc.weight.data.unfiform_(-initrange, initrange)
        self.fc.bias.data.zero_()
    
    def forward(self, text, offsets):
        embedded = self.embedding(text)

        # extract individual documents from text using offsets
        means = []
        for i in range(1, len(offsets)):
            text_tmp = embedded[offsets[i-1]:offsets[i]]
            means.append(text_tmp.mean(0))
        
    
        return self.fc(torch.stack(means))
